
package com.askyria.plugin;

import io.papermc.paper.event.player.AsyncChatEvent;
import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.minimessage.MiniMessage;
import org.bukkit.Bukkit;
import org.bukkit.ChatColor;
import org.bukkit.command.*;
import org.bukkit.configuration.ConfigurationSection;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerJoinEvent;
import org.bukkit.event.player.PlayerQuitEvent;
import org.bukkit.plugin.java.JavaPlugin;

import java.util.*;

public final class AskyriaPlugin extends JavaPlugin implements Listener, CommandExecutor, TabCompleter {
    private final Map<UUID, String> tags = new HashMap<>();
    private final MiniMessage mm = MiniMessage.miniMessage();

    @Override
    public void onEnable() {
        saveDefaultConfig();
        ConfigurationSection section = getConfig().getConfigurationSection("tags");
        if (section != null) {
            for (String key : section.getKeys(false)) {
                try {
                    String value = section.getString(key);
                    if (value != null) tags.put(UUID.fromString(key), value);
                } catch (IllegalArgumentException ignored) {}
            }
        }
        Objects.requireNonNull(getCommand("playertag")).setExecutor(this);
        Objects.requireNonNull(getCommand("playertag")).setTabCompleter(this);
        Bukkit.getPluginManager().registerEvents(this, this);
        for (Player p : Bukkit.getOnlinePlayers()) applyTag(p);
        getLogger().info("Askyria enabled!");
    }

    private void saveTags() {
        getConfig().set("tags", null);
        for (Map.Entry<UUID, String> e : tags.entrySet())
            getConfig().set("tags." + e.getKey(), e.getValue());
        saveConfig();
    }

    private Component prefix(Player p) {
        String value = tags.get(p.getUniqueId());
        if (value == null) return Component.empty();
        return mm.deserialize(value);
    }

    private void applyTag(Player p) {
        Component name = prefix(p).append(Component.text(" ")).append(Component.text(p.getName()));
        if (!tags.containsKey(p.getUniqueId())) name = Component.text(p.getName());
        p.displayName(name);
        p.playerListName(name);
    }

    @EventHandler
    public void onJoin(PlayerJoinEvent e) {
        e.joinMessage(null);
        applyTag(e.getPlayer());
        Bukkit.getScheduler().runTaskLater(this, () -> {
            for (Player p : Bukkit.getOnlinePlayers()) applyTag(p);
        }, 2L);
    }

    @EventHandler
    public void onQuit(PlayerQuitEvent e) {
        e.quitMessage(null);
    }

    @EventHandler
    public void onChat(AsyncChatEvent e) {
        Player p = e.getPlayer();
        Component tag = prefix(p);
        if (!tags.containsKey(p.getUniqueId())) return;
        e.renderer((source, displayName, message, viewer) ->
            tag.append(Component.text(" "))
               .append(source.displayName())
               .append(Component.text(": "))
               .append(message));
    }

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        if (!sender.hasPermission("askyria.admin")) {
            sender.sendMessage(ChatColor.RED + "No permission.");
            return true;
        }
        if (args.length < 2) {
            sender.sendMessage(ChatColor.YELLOW + "Usage: /playertag <player> <color|fade|fade3|remove>");
            return true;
        }

        Player target = Bukkit.getPlayerExact(args[0]);
        if (target == null) {
            sender.sendMessage(ChatColor.RED + "That player must be online.");
            return true;
        }

        String mode = args[1].toLowerCase(Locale.ROOT);
        String value;
        if (mode.equals("remove")) {
            tags.remove(target.getUniqueId());
            saveTags();
            applyTag(target);
            sender.sendMessage(ChatColor.GREEN + "Tag removed.");
            return true;
        } else if (mode.equals("color") && args.length == 3) {
            value = "<" + args[2] + ">[" + args[2].toUpperCase(Locale.ROOT) + "]</" + args[2] + ">";
        } else if (mode.equals("fade") && args.length == 4) {
            value = "<gradient:" + args[2] + ":" + args[3] + ">[" + args[2].toUpperCase(Locale.ROOT) + "]</gradient>";
        } else if (mode.equals("fade3") && args.length == 5) {
            value = "<gradient:" + args[2] + ":" + args[3] + ":" + args[4] + ">[" + args[2].toUpperCase(Locale.ROOT) + "]</gradient>";
        } else {
            sender.sendMessage(ChatColor.RED + "Invalid arguments.");
            return true;
        }

        try {
            mm.deserialize(value);
        } catch (Exception ex) {
            sender.sendMessage(ChatColor.RED + "Invalid color or gradient.");
            return true;
        }

        tags.put(target.getUniqueId(), value);
        saveTags();
        applyTag(target);
        sender.sendMessage(ChatColor.GREEN + "Tag updated for " + target.getName() + ".");
        return true;
    }

    @Override
    public List<String> onTabComplete(CommandSender sender, Command command, String alias, String[] args) {
        if (!sender.hasPermission("askyria.admin")) return Collections.emptyList();
        if (args.length == 1) {
            List<String> names = new ArrayList<>();
            for (Player p : Bukkit.getOnlinePlayers()) names.add(p.getName());
            return names;
        }
        if (args.length == 2) return Arrays.asList("color", "fade", "fade3", "remove");
        if (args.length >= 3) return Arrays.asList("red", "blue", "green", "gold", "aqua", "yellow", "white", "dark_red");
        return Collections.emptyList();
    }
}
